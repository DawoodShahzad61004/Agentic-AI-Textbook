def fan_out_retrievals(state: GraphState):
    """Fan out one retrieve Send per query variant.

    Each Send runs retrieve() in parallel with query=variant.  retrieve() calls
    retrieve_separate() which queries both collections with a single embedding,
    writing to retrieved_document_chunks and retrieved_learned_qa_chunks via the
    Annotated(operator.add) reducers so all variant results accumulate.
    """
    return [
        Send("retrieve", {**state, "query": variant})
        for variant in state["query_variants"]
    ]


def route_user_input(state: GraphState) -> str:

            state.get("retry_count", 0),
        )
        return "retry"
    logger.warning("[ROUTE] INSUFFICIENT but no budget left — using best-effort draft")
    return "generate_answer"

def build_graph():
    graph = StateGraph(GraphState)

    # ── Entry point ───────────────────────────────────────────────────────────
    graph.add_node("user_input", user_input_node)

    # ── Command nodes (each terminates at END) ────────────────────────────────
    graph.add_node("cmd_stats", cmd_stats)
    graph.add_node("cmd_learn", cmd_learn)
    graph.add_node("cmd_bad", cmd_bad)
    graph.add_node("cmd_exit", cmd_exit)

    # ── Query variant generation ──────────────────────────────────────────────
    graph.add_node("generate_query_variants", generate_query_variants)

    # ── Parallel retrieval ────────────────────────────────────────────────────
    graph.add_node("retrieve", retrieve)

    # ── Per-track retrieval validation (parallel) ─────────────────────────────
    graph.add_node("validate_document_retrieval", validate_document_retrieval)
    graph.add_node("validate_learned_qa_retrieval", validate_learned_qa_retrieval)

    # ── Per-track dedup-merge (parallel) ──────────────────────────────────────
    graph.add_node("dedup_merge_documents", dedup_merge_documents)
    graph.add_node("dedup_merge_learned_qa", dedup_merge_learned_qa)

    # ── Per-track dedup-merge validation (parallel, always-present pass-throughs)
    graph.add_node("validate_dedup_merge_documents", validate_dedup_merge_documents)
    graph.add_node("validate_dedup_merge_learned_qa", validate_dedup_merge_learned_qa)

    # ── Document compression track (parallel with learned_qa track) ───────────
    # NAC → DC → LBC, each with an optional validation step.
    # validate_LBC_documents is always present (pass-through when validation is
    # disabled) to give combine_tracks a fixed set of two predecessors.
    graph.add_node("NAC_documents", execute_nac_documents)
    graph.add_node("validate_NAC_documents", validate_nac_documents)
    graph.add_node("DC_documents", execute_dc_documents)
    graph.add_node("validate_DC_documents", validate_dc_documents)
    graph.add_node("LBC_documents", execute_lbc_documents)
    graph.add_node("validate_LBC_documents", validate_lbc_documents)

    # ── Learned-QA compression track (parallel with documents track) ──────────
    # DC → LBC (no NAC), each with an optional validation step.
    # validate_LBC_learned_qa is always present for the same fan-in reason.
    graph.add_node("DC_learned_qa", execute_dc_learned_qa)
    graph.add_node("validate_DC_learned_qa", validate_dc_learned_qa)
    graph.add_node("LBC_learned_qa", execute_lbc_learned_qa)
    graph.add_node("validate_LBC_learned_qa", validate_lbc_learned_qa)

    # ── Track join (fan-in barrier after both compression pipelines complete) ──
    # combine_tracks waits for BOTH validate_LBC_* nodes to finish.
    graph.add_node("combine_tracks", combine_tracks)
    graph.add_node("no_context_answer", no_context_answer)

    # ── Answer generation pipeline ────────────────────────────────────────────
    graph.add_node("generate_draft", generate_draft)
    graph.add_node("check_answer_quality", check_answer_quality)
    graph.add_node("generate_answer", generate_answer)
    graph.add_node("auto_distillation", auto_distillation)

    # ══════════════════════════════════════════════════════════════════════════
    # Edges
    # ══════════════════════════════════════════════════════════════════════════

    graph.add_edge(START, "user_input")
    graph.add_conditional_edges(
        "user_input",
        route_user_input,
        {
            "cmd_stats": "cmd_stats",
            "cmd_learn": "cmd_learn",
            "cmd_bad": "cmd_bad",
            "cmd_exit": "cmd_exit",
            "generate_query_variants": "generate_query_variants",
        },
    )

    graph.add_edge("cmd_stats", END)
    graph.add_edge("cmd_learn", END)
    graph.add_edge("cmd_bad", END)
    # cmd_exit raises SystemExit, no outgoing edge needed.

    # ── Fan-out: one Send per variant → retrieve (all parallel) ──────────────
    graph.add_conditional_edges(
        "generate_query_variants",
        fan_out_retrievals,
        ["retrieve"],
    )

    # ── After retrieve barrier: two validation nodes run in parallel ──────────
    graph.add_edge("retrieve", "validate_document_retrieval")
    graph.add_edge("retrieve", "validate_learned_qa_retrieval")

    # ── Each validation feeds its own dedup-merge (parallel pair) ─────────────
    graph.add_edge("validate_document_retrieval", "dedup_merge_documents")
    graph.add_edge("validate_learned_qa_retrieval", "dedup_merge_learned_qa")

    # ── Each dedup-merge feeds its own validator (parallel pair) ──────────────
    graph.add_edge("dedup_merge_documents", "validate_dedup_merge_documents")
    graph.add_edge("dedup_merge_learned_qa", "validate_dedup_merge_learned_qa")

    # ── Document compression track ────────────────────────────────────────────
    graph.add_edge("validate_dedup_merge_documents", "NAC_documents")
    graph.add_conditional_edges(
        "NAC_documents",
        route_nac_documents_to_validator,
        {"validate_NAC_documents": "validate_NAC_documents", "DC_documents": "DC_documents"},
    )
    graph.add_edge("validate_NAC_documents", "DC_documents")
    graph.add_conditional_edges(
        "DC_documents",
        route_dc_documents_to_validator,
        {"validate_DC_documents": "validate_DC_documents", "LBC_documents": "LBC_documents"},
    )
    graph.add_edge("validate_DC_documents", "LBC_documents")
    graph.add_edge("LBC_documents", "validate_LBC_documents")
    graph.add_edge("validate_LBC_documents", "combine_tracks")

    # ── Learned-QA compression track ─────────────────────────────────────────
    graph.add_edge("validate_dedup_merge_learned_qa", "DC_learned_qa")
    graph.add_conditional_edges(
        "DC_learned_qa",
        route_dc_learned_qa_to_validator,
        {"validate_DC_learned_qa": "validate_DC_learned_qa", "LBC_learned_qa": "LBC_learned_qa"},
    )
    graph.add_edge("validate_DC_learned_qa", "LBC_learned_qa")
    graph.add_edge("LBC_learned_qa", "validate_LBC_learned_qa")
    graph.add_edge("validate_LBC_learned_qa", "combine_tracks")

    # ── After combine: route to draft/answer pipeline, retry, or no_context_answer
    graph.add_conditional_edges(
        "combine_tracks",
        route_after_combine,
        {
            "generate_draft": "generate_draft",
            "generate_answer": "generate_answer",
            "retry": "generate_query_variants",
            "no_context_answer": "no_context_answer",
        },
    )
    graph.add_edge("no_context_answer", END)

    # ── Draft → quality check → generate_answer ───────────────────────────────
    graph.add_conditional_edges(
        "generate_draft",
        route_after_generate_draft,
        {"check_answer_quality": "check_answer_quality", "generate_answer": "generate_answer"},
    )
    graph.add_conditional_edges(
        "check_answer_quality",
        route_after_quality_check,
        {"generate_answer": "generate_answer", "retry": "generate_query_variants"},
    )

    # ── Final answer → optional distillation → END ────────────────────────────
    graph.add_conditional_edges(
        "generate_answer",
        route_after_generate_answer,
        {"auto_distillation": "auto_distillation", END: END},
    )
    graph.add_edge("auto_distillation", END)

    return graph.compile()

def _load_failed_variants() -> dict:
    if not FAILED_VARIANTS_PATH.exists():
        return {}
    try:
        return json.loads(FAILED_VARIANTS_PATH.read_text(encoding="utf-8"))

        return {}


def _save_failed_variants(store: dict) -> None:
    FAILED_VARIANTS_PATH.parent.mkdir(parents=True, exist_ok=True)
    FAILED_VARIANTS_PATH.write_text(
        json.dumps(store, indent=2, ensure_ascii=False), encoding="utf-8"
    )

@app.post("/query")
async def query(req: QueryRequest):
    if not req.query.strip():
        raise HTTPException(status_code=400, detail="Query must not be empty.")

    norm = req.query.lower().strip()
    failed_store = _ctx["failed_variants_store"]
    blocked: list[str] = list(failed_store.get(norm, []))

    feedback_store: FeedbackStore = _ctx["feedback_store"]
    prior_thumbdowns = feedback_store.find_thumbdowns_for_query(req.query)

    existing_blocked: set[str] = set(blocked)
    for td in prior_thumbdowns:
        for v in (td.get("variants") or []):
            vq = (v.get("query") or "").strip()
            if vq and vq not in existing_blocked:
                blocked.append(vq)
                existing_blocked.add(vq)

    async with _query_lock:
        try:
            result, newly_failed = await asyncio.to_thread(
                run_agent,
                req.query,
                _ctx["llm"],
                _ctx["merge_llm"],
                _ctx["judge_llm"],
                _ctx["tool_schemas"],
                _ctx["callables"],
                _ctx["check_answer_quality"],
                _ctx["retriever"],
                _ctx["agent_state"],
                blocked,
                prior_thumbdowns,
            )
        except LLMRateLimitAbortError as e:
            raise HTTPException(
                status_code=429,
                detail=(
                    f"Rate limit hit. Required backoff: {e.delay:.0f}s. "
                    f"Please retry after {e.delay:.0f} seconds."
                ),
                headers={"Retry-After": str(int(e.delay))},
            )

    _ctx["last_result"] = result

    if newly_failed:
        existing = set(failed_store.get(norm, []))
        fresh = [v for v in newly_failed if v not in existing]
        if fresh:
            failed_store.setdefault(norm, []).extend(fresh)
            _save_failed_variants(failed_store)

    feedback_store.log(
        query=req.query,
        answer=result["answer"],
        sources=result["sources"],
        quality=result["quality"],
        document_retrieved_chunks=result.get("document_chunks", []),
        learned_qa_retrieved_chunks=result.get("learned_qa_chunks", []),
    )

    if (
        ENABLE_AUTO_DISTILLATION
        and result["quality"] == "OK"
        and _ctx["self_learner"].should_learn()
    ):
        added = await asyncio.to_thread(_ctx["self_learner"].run_distillation)
        logger.info("[Self-Learning] %d new QA pair(s) added to knowledge base", added)

    return result


@app.post("/feedback/bad")
async def feedback_bad(req: FeedbackRequest):

    variants = last.get("variants", []) if last else []
    ok = feedback_store.mark_last_bad(feedback=req.feedback, variants=variants)

In app_workflow, LangGraph's graph structure is the loop — the retry edge cycles back through retrieval, and LLM_RESPONSE_RETRY_LIMIT (already used in routes.py:66,91) caps how many times that retry path can fire. MAX_TOTAL_RETRIEVALS — capped how many total retrieval calls the agent could make across all tool calls. Exponential backoff — capped:


delay = min(max_seconds, base_seconds * 2^(attempt-1))
Base: 1.0s, so delays are 1s → 2s → 4s → ...
Hard cap: 30.0s — the exponential never exceeds this

delay += random.uniform(0.0, jitter_seconds)
Max jitter added: 0.25s (full-jitter, uniform from 0 to 0.25)

def _parse(output: str) -> tuple[float, str]:
    elapsed = 0.0
    for line in output.splitlines():
        if line.startswith(_EL):
            try:
                elapsed = float(line[len(_EL):])
            except ValueError:
                pass

    answer = ""
    if _AS in output and _AE in output:
        s = output.index(_AS) + len(_AS) + 1   # +1 to skip the trailing newline
        e = output.index(_AE)
        answer = output[s:e].strip()

    return elapsed, answer


def _run_one(mode: str, run_num: int, label: str) -> tuple[float, str]:
    print(f"  [{label}] Run {run_num}/{RUNS} … ", end="", flush=True)
    proc = subprocess.run(
        [sys.executable, str(ROOT / "benchmark.py"), f"--_mode={mode}", QUERY],
        capture_output=True,
        text=True,
    )
    if proc.returncode != 0:
        err = (proc.stderr or "")[:500].strip()
        print(f"FAILED (rc={proc.returncode})")
        if err:
            # Print the last 3 lines of stderr for quick diagnosis
            tail = "\n".join(err.splitlines()[-3:])
            print(f"    stderr tail: {tail}")
        return 0.0, f"[ERROR rc={proc.returncode}]"

    elapsed, answer = _parse(proc.stdout)
    print(f"done in {elapsed:.2f} s")
    return elapsed, answer

def _fmt_answer(answer: str, indent: int = 4) -> str:
    """Indent every line of a potentially multi-line answer."""
    pad = " " * indent
    return "\n".join(f"{pad}{line}" for line in answer.splitlines()) if answer else f"{' ' * indent}(no answer)"

What do u mean by pipeline workers? How r they different from threads?

This is an excellent question because many engineers initially think:

"A worker is just a thread."

Not necessarily.

A pipeline worker is an architectural concept.

A thread is an execution mechanism provided by the operating system/runtime.

They solve different problems.

1. What is a thread?

A thread is simply a unit of execution.

Example:

def run_pipeline(query):
    ...

thread1 = Thread(target=run_pipeline, args=("What is autism?",))
thread2 = Thread(target=run_pipeline, args=("What is ADHD?",))

The OS schedules them.

Think:

CPU
 ├─ Thread 1
 ├─ Thread 2
 └─ Thread 3

A thread knows nothing about:

RAG
LangGraph
Pipelines
Workers
Requests

It just executes code.

2. What is a pipeline worker?

A worker is usually:

class PipelineWorker:
    def __init__(self):
        self.pipeline = ...
        self.retriever = ...
        self.agent_state = ...
        self.lock = ...

It is an object that owns resources.

Example:

Worker #1
 ├─ LangGraph app
 ├─ Retriever
 ├─ Agent state
 └─ Tool set

Worker #2
 ├─ LangGraph app
 ├─ Retriever
 ├─ Agent state
 └─ Tool set

Requests are assigned to workers.

Analogy

Imagine a restaurant.

Thread

A thread is:

A waiter

who can do work.

Worker

A worker is:

A complete kitchen station

containing:

Chef
Oven
Tools
Ingredients
Counter

The worker owns resources.

The thread merely performs actions.

Example 1: One pipeline, many threads

Suppose you have:

rag_app = build_graph()

and all requests use it.

Request A -> rag_app.invoke()

Request B -> rag_app.invoke()

Request C -> rag_app.invoke()

Multiple threads are executing:

rag_app.invoke()

at the same time.

Problem:

same graph
same state
same objects
same memory

Possible race conditions.

This is why many people add:

_query_lock
Example 2: Worker pool

Instead:

worker1 = PipelineWorker()
worker2 = PipelineWorker()
worker3 = PipelineWorker()

Now:

Request A -> Worker 1
Request B -> Worker 2
Request C -> Worker 3

Each worker has:

its own graph
its own state
its own tools

Now concurrency becomes much safer.

This is where many people get confused.

Worker:

Worker 1

or:

ThreadPoolExecutor

or:

ProcessPoolExecutor

So:

Worker != Thread

A worker may use:

1 thread
many threads
1 process
many processes

depending on implementation.
