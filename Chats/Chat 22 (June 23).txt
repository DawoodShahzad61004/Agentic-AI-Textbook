from __future__ import annotations

import json
import threading
from pathlib import Path
from typing import Optional

CATEGORIES = [
    "Sub-Query Generation",
    "Total DB Retrieval Time",
    "Total DB Retrieval Validation Time",
    "Total Merge Time for Retrieved Chunks",
    "Total Validation Time for Merged Chunks",
    "Compression",
    "Draft Generation",
    "CAQ",
    "Final Generation",
]


class TimingTracker:
    def __init__(self) -> None:
        self._lock = threading.Lock()
        self._data: dict[str, list[float]] = {cat: [] for cat in CATEGORIES}
        self._json_path: Optional[Path] = None

    def set_json_path(self, path: Path) -> None:
        with self._lock:
            self._json_path = path
            self._flush_unlocked()

    def record(self, category: str, duration: float) -> None:
        with self._lock:
            if category in self._data:
                self._data[category].append(round(duration, 4))
            self._flush_unlocked()

    def _flush_unlocked(self) -> None:
        if self._json_path is None:
            return
        try:
            self._json_path.parent.mkdir(parents=True, exist_ok=True)
            self._json_path.write_text(
                json.dumps(self._data, indent=2),
                encoding="utf-8",
            )
        except Exception:
            pass

"""Singleton timing tracker that records per-phase durations to a JSON file.

Initialized by logger_config.setup_logging(); all other modules call
record() or record_llm() without needing the file path.
"""
from __future__ import annotations

import json
import threading
from pathlib import Path

CATEGORIES = (
    "Sub-Query Generation",
    "Total DB Retrieval Time",
    "Total DB Retrieval Validation Time",
    "Total Merge Time for Retrieved Chunks",
    "Total Validation Time for Merged Chunks",
    "Compression",
    "Draft Generation",
    "CAQ",
    "Final Generation",
)

# Map llm_invoke caller_tag → timing category.
# Tags not listed here are silently ignored by record_llm().
_LLM_TAG_TO_CATEGORY: dict[str, str] = {
    "AGENT-RETRIEVE": "Sub-Query Generation",
    "AGENT-DRAFT":    "Draft Generation",
    "AGENT-FINAL":    "Final Generation",
    "CAQ-JUDGE":      "CAQ",
    "CAQ-JUDGE-TD":   "CAQ",
}

_data: dict[str, list[float]] = {c: [] for c in CATEGORIES}
_json_path: Path | None = None
_lock = threading.Lock()


def initialize(json_path: Path) -> None:
    """Set the output file and reset all lists. Called once by setup_logging()."""
    global _data, _json_path
    with _lock:
        _json_path = json_path
        _data = {c: [] for c in CATEGORIES}
        _write()


def record(category: str, duration: float) -> None:
    """Append a duration (seconds) to the given category list and flush to disk."""
    with _lock:
        if category in _data:
            _data[category].append(round(duration, 4))
            _write()


def record_llm(caller_tag: str, duration: float) -> None:
    """Resolve caller_tag to a category and record; no-op for unmapped tags."""
    category = _LLM_TAG_TO_CATEGORY.get(caller_tag)
    if category:
        record(category, duration)


def _write() -> None:
    if _json_path is None:
        return
    try:
        _json_path.write_text(json.dumps(_data, indent=2), encoding="utf-8")
    except OSError:
        pass

import requests
import time
import sys
import os

QUERIES = [
    "How does frequency affect motor speed in an ASD?",
    "What are the benefits of using adjustable speed drives?",
    "How do modern ASDs differ from older mechanical speed control methods?",
    "What role do adjustable speed drives play in industrial energy efficiency?",
    "How does an ASD control motor acceleration and deceleration?",
    "Give me an overview of adjustable speed drives",
    "Can you describe what adjustable speed drives are?",
]

BASE_URL = "http://localhost:8001"
API_SCRIPT = os.path.join(os.path.dirname(os.path.abspath(__file__)), "app", "api.py")


def start_api():
    print("Starting app/api.py ...")
    proc = subprocess.Popen([sys.executable, API_SCRIPT])
    print("Pausing 2 minutes for server startup ...")
    time.sleep(120)
    return proc


def post_query(query):
    print(f"  POST /query  |  {query!r}")
    resp = requests.post(f"{BASE_URL}/query", json={"query": query}, timeout=300)
    print(f"  [{resp.status_code}] {resp.text[:300]}")
    return resp


def quit_server():
    print("  POST /quit")
    try:
        resp = requests.post(f"{BASE_URL}/quit", timeout=30)
        print(f"  [{resp.status_code}]")
    except Exception as exc:
        print(f"  quit error (expected if server already stopped): {exc}")


def run_steps_1_to_3(query):
    post_query(query)
    quit_server()
    print("Pausing 5 minutes ...")
    time.sleep(300)


for idx, query in enumerate(QUERIES):
    print(f"\n{'=' * 60}")

    # Step 5: pause 10 minutes before the next query (skip after last)

        print("Pausing 10 minutes before next query ...")
        time.sleep(600)

print("\nAll queries completed.")
