What changes and why
Your current pipeline in query.py is a straight line: embed → retrieve → prompt → answer. To make it agentic, you give the LLM tools and let it decide when and how to use them. The LLM can call retrieve, check if the results are good, call it again with a different query, or decide it has enough to answer. This is the core loop.

# tools.py
from retriever import RAGRetriever
from embedding_manager import EmbeddingManager
from vector_store import VectorStore

def make_tools(retriever: RAGRetriever):
    """Return tool definitions + callables for the agent."""

    def retrieve_documents(query: str, top_k: int = 5) -> str:
        """Search the vector store for relevant chunks."""
        docs = retriever.retrieve(query, top_k=top_k, score_threshold=0.2)
        if not docs:
            return "No relevant documents found for this query."
        return "\n\n".join(
            f"[Source: {d['metadata'].get('source','?')}, score: {d['similarity_score']:.3f}]\n{d['content']}"
            for d in docs
        )

    def check_answer_quality(answer: str, context: str) -> str:
        """Heuristic: flag if answer looks like a hallucination or low confidence."""
        red_flags = ["i don't know", "i cannot", "no information", "not mentioned"]
        if any(f in answer.lower() for f in red_flags):
            return "INSUFFICIENT — agent should retry with a different query."
        return "OK"

    # LangChain-style tool schema (used when calling the LLM)
    tool_schemas = [
        {
            "type": "function",
            "function": {
                "name": "retrieve_documents",
                "description": "Search the knowledge base for chunks relevant to a query. Call this first, and again with rephrased queries if results are poor.",
                "parameters": {
                    "type": "object",
                    "properties": {
                        "query": {"type": "string", "description": "The search query."},
                        "top_k": {"type": "integer", "description": "Number of results to return.", "default": 5},
                    },
                    "required": ["query"],
                },
            },
        },
        {
            "type": "function",
            "function": {
                "name": "check_answer_quality",
                "description": "Check whether a drafted answer is grounded. Returns OK or INSUFFICIENT.",
                "parameters": {
                    "type": "object",
                    "properties": {
                        "answer": {"type": "string"},
                        "context": {"type": "string"},
                    },
                    "required": ["answer", "context"],
                },
            },
        },
    ]

    callables = {
        "retrieve_documents": retrieve_documents,
        "check_answer_quality": check_answer_quality,
    }

    return tool_schemas, callables
Step 2 — Replace query.py with an agent loop
The key change is switching from a single llm.invoke(prompt) call to a while loop that keeps running until the LLM stops calling tools:

python
# agent_query.py
import json, os, argparse
from dotenv import load_dotenv
from langchain_groq import ChatGroq

from embedding_manager import EmbeddingManager
from vector_store import VectorStore
from retriever import RAGRetriever
from tools import make_tools

VECTOR_STORE_PATH = "../data/vector_store"
COLLECTION_NAME = "documents"
MAX_ITERATIONS = 6  # safety ceiling — agent won't loop forever

SYSTEM_PROMPT = """You are a research assistant with access to a knowledge base.

When given a question:
1. Call retrieve_documents with the most relevant search query.
2. Read the results carefully.
3. If results are sparse or off-topic, call retrieve_documents again with a REPHRASED query.
4. Draft your answer, then call check_answer_quality to verify it.
5. If quality is INSUFFICIENT, try once more with a different angle.
6. Return your final answer with source citations.

Never make up facts. If the knowledge base lacks the information, say so clearly."""


def run_agent(query: str, llm, tool_schemas: list, callables: dict) -> dict:
    messages = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user", "content": query},
    ]
    iterations = 0
    retrieved_sources = []

    while iterations < MAX_ITERATIONS:
        iterations += 1

        # Ask the LLM what to do next
        response = llm.invoke(messages, tools=tool_schemas)

        # If no tool call → agent is done, return the answer
        if not response.tool_calls:
            return {
                "answer": response.content,
                "sources": retrieved_sources,
                "iterations": iterations,
            }

        # Execute each tool the agent requested
        messages.append(response)  # add assistant message with tool_calls
        for tool_call in response.tool_calls:
            name = tool_call["name"]
            args = tool_call["args"]
            print(f"  [agent] calling tool: {name}({args})")

            result = callables[name](**args)

            # Track sources for final output
            if name == "retrieve_documents":
                retrieved_sources.append({"query": args["query"], "preview": result[:200]})

            # Feed result back as a tool message
            messages.append({
                "role": "tool",
                "tool_call_id": tool_call["id"],
                "content": result,
            })

    return {"answer": "Max iterations reached without a confident answer.", "sources": retrieved_sources, "iterations": iterations}


def main():
    load_dotenv()
    embedding_manager = EmbeddingManager()
    vector_store = VectorStore(collection_name=COLLECTION_NAME, persist_directory=VECTOR_STORE_PATH)

    if vector_store.collection.count() == 0:
        print("Vector store is empty. Run ingest.py first.")
        return

    retriever = RAGRetriever(vector_store, embedding_manager)
    llm = ChatGroq(
        api_key=os.getenv("GROQ_API_KEY"),
        model_name="llama-3.1-8b-instant",
        temperature=0.1,
        max_tokens=2048,
    )

    tool_schemas, callables = make_tools(retriever)

    print("\n=== Agentic RAG — Query Interface ===")
    print("The agent will decide when to retrieve, re-retrieve, and verify.\n")

    while True:
        query = input("Your question: ").strip()
        if not query or query.lower() in {"quit", "exit"}:
            break

        result = run_agent(query, llm, tool_schemas, callables)

        print(f"\nANSWER:\n{result['answer']}")
        print(f"\nIterations used: {result['iterations']}")
        print("Sources searched:")
        for s in result["sources"]:
            print(f"  query: '{s['query']}'\n  preview: {s['preview'][:100]}...")
        print("=" * 50)


if __name__ == "__main__":
    main()

# Add to retriever.py
def retrieve_multi_query(self, queries: list[str], top_k: int = 5) -> list[dict]:
    """Retrieve for multiple query variants and deduplicate by content hash."""
    seen, results = set(), []
    for q in queries:
        for doc in self.retrieve(q, top_k=top_k):
            key = hash(doc["content"][:100])
            if key not in seen:
                seen.add(key)
                results.append(doc)
    results.sort(key=lambda d: d["similarity_score"], reverse=True)
    return results[:top_k * 2]

No — and this is an important distinction to understand.

What you've built (even after the agentic upgrade) is stateless by design. Every conversation starts fresh. The agent gets smarter within a single session by retrying and self-correcting, but once the conversation ends, nothing is remembered. The next user starts from scratch.

Why agentic ≠ self-learning:

The LLM's weights are frozen — Groq is serving you a fixed model that Anthropic/Meta trained. Your code never touches those weights.
The vector store only changes when you explicitly run ingest.py. The agent can read from it but never writes to it.
MAX_ITERATIONS reasoning is ephemeral — it lives in the messages list in RAM and disappears when the process exits.
What would actually give you self-improvement?
There are three real mechanisms, in increasing complexity:

1. Memory injection (easiest — hours of work) Save useful Q&A pairs back into the vector store after each session. Next time someone asks a similar question, the agent retrieves a previously-verified answer as additional context. This is pseudo-learning — the knowledge base grows, not the model.

python
# After a successful agent run, write back to the store
if result["confidence"] > 0.85:
    vector_store.add_qa_pair(query, result["answer"])
2. Fine-tuning on accumulated interactions (moderate — requires infra) Log every query + high-rated answer pair to a dataset. Periodically fine-tune a smaller open-source model (like Mistral 7B via Unsloth or Axolotl) on that dataset. Swap it in as your LLM. This is real weight-level learning, but it's a batch process — not continuous.

3. Reinforcement learning from user feedback (hardest — research-grade) Capture thumbs up/down signals per answer, use them as a reward signal, and run RLHF or DPO training loops. This is what OpenAI and Anthropic do at scale. For a single-app context it's overkill and expensive.

The honest recommendation for your use case
The most practical path is option 1 + structured feedback logging:

Write good answers back to the vector store (memory injection)
Log every interaction to a file or database
Periodically review logs and add high-quality exchanges as new documents via ingest.py
That's a human-in-the-loop improvement cycle, which is actually more reliable than fully automated self-learning for domain-specific apps
True continuous self-learning without human oversight also carries a real risk: the model can learn from its own mistakes and reinforce bad answers. Memory injection with a confidence threshold is safer and nearly as effective for most RAG use cases.

This single line is where the "agentic" magic actually happens. Let me break it down.

What llm.invoke() normally does
In your original query.py, you called:

python
response = llm.invoke(prompt)
That's a simple text-in, text-out call. The LLM reads the prompt and writes an answer. Done.

What changes when you pass tools=
python
response = llm.invoke(messages, tools=tool_schemas)
You're now sending the LLM two things:

messages — the full conversation history, including the system prompt, user question, and any previous tool results. It's a list, not a string, because the agent needs context from prior iterations of the loop.

tools=tool_schemas — a JSON description of every function the agent is allowed to call. This tells the LLM "you don't have to answer right now — you can request data first."

Under the hood, LangChain serializes tool_schemas and appends it to the request sent to Groq. The model was trained to recognize this format and respond differently when it sees it.

The two possible responses
After this call, response can be one of two shapes:

Shape A — the agent wants to call a tool:

python
response.content    # "" or None — no answer yet
response.tool_calls # [{"name": "retrieve_documents", "args": {"query": "ASD symptoms"}, "id": "call_abc"}]
Shape B — the agent is ready to answer:

python
response.content    # "ASD is characterized by..."
response.tool_calls # [] — empty, agent is done
This is why your loop condition is if not response.tool_calls — that's how you detect when the agent has finished reasoning and is giving you its final answer.

What happens inside the model
The LLM doesn't actually execute your Python function. It just decides it wants to — it outputs a structured JSON object saying "call this function with these arguments." Your loop in run_agent is what actually executes the function and feeds the result back. The model never touches your code directly.

LLM says:  "call retrieve_documents(query='ASD symptoms')"
Your loop: actually calls it, gets the chunks back
Your loop: appends result to messages
LLM says:  "now I have enough — here's the answer"
This back-and-forth is the entire agentic pattern. The llm.invoke() line is just one step in that cycle — it gets called once per iteration of your while loop.

Claude responded: Great question.
Great question. response.tool_calls is not something you build — it's produced automatically by LangChain when it processes the LLM's raw response.

Where it comes from
When you call llm.invoke(messages, tools=tool_schemas), Groq returns a raw HTTP response that looks like this under the hood:

json
{
  "choices": [{
    "message": {
      "role": "assistant",
      "content": null,
      "tool_calls": [
        {
          "id": "call_abc123",
          "type": "function",
          "function": {
            "name": "retrieve_documents",
            "arguments": "{\"query\": \"ASD symptoms\", \"top_k\": 5}"
          }
        }
      ]
    }
  }]
}
LangChain's ChatGroq class receives this, parses it, and wraps it into its own AIMessage object. It does two things in that parsing step:

Deserializes "arguments" from a raw JSON string into a proper Python dict
Exposes everything neatly as response.tool_calls
So by the time the object reaches your code, response.tool_calls is already a clean Python list.

The exact structure
python
response.tool_calls = [
    {
        "id":   "call_abc123",        # str — unique ID for this specific call
        "name": "retrieve_documents", # str — matches a key in your callables dict
        "args": {                     # dict — already parsed, ready for **unpacking
            "query": "ASD symptoms",
            "top_k": 5
        }
    }
]
Three keys, always. id, name, and args. This is LangChain's standardized format — it's the same structure regardless of whether you're using Groq, OpenAI, or Anthropic as your backend.

Why args is already a dict
Notice in the raw Groq response, arguments is a JSON string — "{\"query\": \"ASD symptoms\"}". LangChain automatically calls json.loads() on it before giving you the object. That's why **args works directly — if LangChain hadn't done that, you'd get a TypeError trying to unpack a string.

Where tool_call_id connects
You might notice later in the code you use tool_call["id"] when feeding results back:

python
messages.append({
    "role": "tool",
    "tool_call_id": tool_call["id"],  # ← "call_abc123"
    "content": result,
})
That ID is how the LLM matches each result to the request that triggered it. If the agent called two tools simultaneously, the IDs ensure result A goes back to request A, not request B.

What your system prompt said

What the LLM actually did
Instead of being obedient and sequential, it looked at the user question, reasoned about it internally, and decided "I'll just try all likely phrasings at once right now" — and returned all 5 tool calls in a single response.

This is called parallel tool calling and it's a capability baked into the model itself during training. Groq's hosted Llama 3.1 was trained to recognize when multiple tool calls are independent of each other and can be batched. Since rephrased retrieval queries don't depend on each other's results, the model correctly identified it could fire them all simultaneously.

Where in the code this is allowed
In llm.invoke(), you pass tools=tool_schemas but you never set any constraint like tool_choice="any" or parallel_tool_calls=False. So the model has full freedom over:

which tools to call
how many to call at once
what arguments to pass them
The only constraint you imposed was MAX_ITERATIONS = 6, which limits loop ticks — not tool calls per tick. So in one iteration, the model can call as many tools as it wants.

Fix 2 — Hard limit tool calls per iteration in code
Even with prompt instructions, a small model like llama-3.1-8b-instant can ignore them. Enforce it in run_agent:

MAX_ITERATIONS = 6
MAX_TOOL_CALLS_PER_ITERATION = 5   # ← add this
MAX_TOTAL_RETRIEVALS = 5           # ← and this
Then enforce it inside the loop:

python
def run_agent(query: str, llm, tool_schemas: list, callables: dict) -> dict:
    messages = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user", "content": query},
    ]
    iterations = 0
    retrieved_sources = []
    total_retrievals = 0          # ← track across all iterations

    while iterations < MAX_ITERATIONS:
        iterations += 1

        response = llm.invoke(messages, tools=tool_schemas)
        print(f"\n[Iteration {iterations}] — tools requested: {[t['name'] for t in response.tool_calls] if response.tool_calls else 'none (final answer)'}")

        if not response.tool_calls:
            return {
                "answer": response.content,
                "sources": retrieved_sources,
                "iterations": iterations,
            }

        # ← Slice off excess tool calls before processing
        tool_calls = response.tool_calls[:MAX_TOOL_CALLS_PER_ITERATION]

        messages.append(response)
        last_context = ""

        for tool_call in tool_calls:
            name = tool_call["name"]
            args = tool_call["args"]

            if name == "retrieve_documents":
                # ← Stop retrieving if we've hit the global cap
                if total_retrievals >= MAX_TOTAL_RETRIEVALS:
                    print(f"  → skipping {name} — retrieval cap reached")
                    messages.append({
                        "role": "tool",
                        "tool_call_id": tool_call["id"],
                        "content": "Retrieval limit reached. Use what you have to answer.",
                    })
                    continue

                result = callables[name](**args)
                last_context = result
                total_retrievals += 1
                retrieved_sources.append({"query": args["query"], "preview": result[:200]})

            elif name == "check_answer_quality":
                args["context"] = last_context
                result = callables[name](**args)
                print(f"     quality check: {result}")

            else:
                result = callables[name](**args)

            messages.append({
                "role": "tool",
                "tool_call_id": tool_call["id"],
                "content": result,
            })

    return {"answer": "Max iterations reached without a confident answer.", "sources": retrieved_sources, "iterations": iterations}
