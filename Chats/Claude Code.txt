def _load_pdf(p: Path) -> list[Document]:
    import fitz
    docs = []
    with fitz.open(str(p)) as pdf:
        for i, page in enumerate(pdf):
            text = page.get_text()
            if text.strip():
                docs.append(Document(page_content=text, metadata={"source": str(p), "page": i}))
    return docs


def _load_txt(p: Path) -> list[Document]:
    text = p.read_text(encoding="utf-8", errors="ignore")
    return [Document(page_content=text, metadata={"source": str(p)})]


def _load_docx(p: Path) -> list[Document]:
    import docx
    doc = docx.Document(str(p))
    text = "\n".join(para.text for para in doc.paragraphs if para.text.strip())
    return [Document(page_content=text, metadata={"source": str(p)})]


def _load_doc(p: Path) -> list[Document]:
    # .doc is the legacy Word binary format; python-docx does not support it.
    # Convert the file to .docx with LibreOffice or Word first for best results.
    try:
        import docx2txt
        text = docx2txt.process(str(p))
        return [Document(page_content=text, metadata={"source": str(p)})]
    except ImportError:
        print(f"Skipping {p.name}: .doc support requires 'docx2txt' (pip install docx2txt)")
        return []


def _load_html(p: Path) -> list[Document]:
    from html.parser import HTMLParser

    class _Extractor(HTMLParser):
        def __init__(self):
            super().__init__()
            self._parts: list[str] = []
            self._skip = False

        def handle_starttag(self, tag, attrs):
            if tag in {"script", "style"}:
                self._skip = True

        def handle_endtag(self, tag):
            if tag in {"script", "style"}:
                self._skip = False

        def handle_data(self, data):
            if not self._skip and data.strip():
                self._parts.append(data.strip())

    extractor = _Extractor()
    extractor.feed(p.read_text(encoding="utf-8", errors="ignore"))
    text = "\n".join(extractor._parts)
    return [Document(page_content=text, metadata={"source": str(p)})]


def _load_json(p: Path) -> list[Document]:
    data = json.loads(p.read_text(encoding="utf-8"))
    docs = []
    if isinstance(data, list):
        for i, record in enumerate(data):
            if isinstance(record, dict):
                text = "\n".join(f"{k}: {v}" for k, v in record.items() if v not in (None, ""))
            else:
                text = str(record)
            if text.strip():
                docs.append(Document(page_content=text, metadata={"source": str(p), "record": i}))
    else:
        docs.append(Document(page_content=json.dumps(data, ensure_ascii=False, indent=2),
                             metadata={"source": str(p)}))
    return docs

    for fp in file_paths:

        ext = p.suffix.lower()

        if ext == ".csv":

                df = df.fillna("")
                records = df.to_dict(orient="records")
                json_path = p.with_suffix(".json")
                json_path.write_text(

                    encoding="utf-8",

                p = json_path
                ext = ".json"
            except Exception as e:

                continue

        loader_fn = _LOADERS.get(ext)
        if loader_fn is None:
            print(f"No loader for {p.name}, skipping")
            continue

        try:
            docs = loader_fn(p)
            all_docs.extend(docs)
        except Exception as e:
            print(f"Failed to load {p.name}: {e}")

    print(f"Loaded {len(all_docs)} document pages/sections")
    return all_docs


def split_documents(documents: list[Document]) -> list[Document]:

        chunk_overlap=CHUNK_OVERLAP,
        length_function=len,

    )
    chunks = splitter.split_documents(documents)

def _serialize_messages(messages: list) -> str:
    """Convert the mixed list of dicts and LangChain AIMessage objects to a readable string."""
    lines = []
    for i, m in enumerate(messages):
        if isinstance(m, dict):
            role    = m.get("role", "?")
            content = str(m.get("content", ""))
            tid     = m.get("tool_call_id", "")
            header  = f"  [{i}] role={role}" + (f"  tool_call_id={tid}" if tid else "")
            lines.append(header)
            lines.append(f"      content: {content[:600]}{'...' if len(content) > 600 else ''}")
        else:
            # LangChain AIMessage
            content    = str(getattr(m, "content", ""))
            tool_calls = getattr(m, "tool_calls", []) or []
            lines.append(f"  [{i}] role=assistant")
            lines.append(f"      content: {content[:600]}{'...' if len(content) > 600 else ''}")
            for tc in tool_calls:
                lines.append(f"      tool_call → {tc['name']}  args={json.dumps(tc['args'])}")
        lines.append("")
    return "\n".join(lines)

def _stable_id(text: str) -> str:
    return hashlib.sha256(text.encode()).hexdigest()[:16]


class SelfLearner:
    def __init__(
        self,
        embedding_manager: EmbeddingManager,
        llm: ChatGroq,
        feedback_store: FeedbackStore,
        vector_store_path: str = VECTOR_STORE_PATH,
        learn_every_n: int = 5,
    ):
        self.embedding_manager = embedding_manager
        self.llm = llm
        self.feedback_store = feedback_store
        self.learn_every_n = learn_every_n

        self.client = chromadb.PersistentClient(path=vector_store_path)
        self.collection = self.client.get_or_create_collection(
            name=LEARNED_COLLECTION,
            metadata={"description": "Distilled QA pairs from verified interactions."},

    def should_learn(self) -> bool:

FeedbackStore manages two files that together power the self-learning loop:

interactions.jsonl — an append-only log. Every query/answer pair is written here via log() with a quality label: OK, INSUFFICIENT, or USER_THUMBSDOWN. The read methods (load_all, load_good, count_good) let other parts of the system query this history — e.g. to pull good examples for distillation.

failed_variants.json — a blocklist used by the agent at query time. It maps a normalized query (lowercased) to a list of known-bad answers for that query, so the agent can avoid regenerating the same bad response.

mark_last_bad() is the bridge between the two. When the user gives a thumbs-down:

It rewrites the last record in interactions.jsonl from whatever quality it had to USER_THUMBSDOWN.
It then extracts that record's query and answer, and appends the answer to failed_variants.json under the normalized query key — but only if that exact answer isn't already listed there (deduplication).
So a thumbs-down now has immediate effect: the next time someone asks the same question, the agent will see the blocked variants and won't produce the same bad answer again.
